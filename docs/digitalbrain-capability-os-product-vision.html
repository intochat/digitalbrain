One bounded C# implementation file · digest 8d4c…a12f

public sealed class CompanyResearchFeature
{
    public async Task<FeatureResult> RunAsync(
        CompanyResearchInput input,
        IFeatureContext context)
    {
        var subject = await context.ResolveCompanyAsync(input.Name);
        var sources = await context.Research.SearchAsync(subject);
        var brief = await context.Compose.AttributedBriefAsync(sources);

        return FeatureResult.TextArtifact(
            $"{subject.Slug}-research.txt",
            brief);
    }
}
