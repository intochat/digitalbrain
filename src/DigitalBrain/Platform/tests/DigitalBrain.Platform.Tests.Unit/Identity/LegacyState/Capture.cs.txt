using DigitalBrain.Sdk.Identity;
using DigitalBrain.Sdk.Identity.Directory;
using DigitalBrain.Sdk.Identity.Grants;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Orleans.Hosting;
using Orleans.Serialization;
using Orleans.Storage;

using var host = new HostBuilder().UseOrleans(silo => silo.UseLocalhostClustering().AddMemoryGrainStorage("Default")).Build();
var binary = new OrleansGrainStorageSerializer(host.Services.GetRequiredService<Serializer>());
var json = new JsonGrainStorageSerializer(host.Services.GetRequiredService<OrleansJsonSerializer>());
var at = DateTimeOffset.Parse("2026-09-30T12:00:00Z");
var directory = new IdentityDirectoryState
{
    Accounts = [new() { AccountId = "legacy-account", Name = "Legacy", OwnerPrincipalId = "alice", CreatedAt = at }],
    Members = [new() { AccountId = "legacy-account", BrainId = "legacy-brain", PrincipalId = "alice", DisplayName = "Alice", Role = MemberRole.Owner, JoinedAt = at }],
    Invitations = [new() { Code = "legacy-invite", AccountId = "legacy-account", WorkspaceId = "legacy-brain", Role = MemberRole.Member, Email = "invite@example.test", CreatedAt = at }],
    PasswordHashes = new() { ["alice"] = new Microsoft.AspNetCore.Identity.PasswordHasher<string>().HashPassword("alice", "legacy-password") },
};
var grants = new GrantStoreState
{
    Grants = [new() { AppId = "legacy-app", SemanticTypeId = "person.email", WorkspaceId = "legacy-brain", Mode = GrantMode.ThisChat, ConversationId = "legacy-chat", GrantedAt = at }],
};
System.IO.Directory.CreateDirectory(args[0]);
File.WriteAllBytes(Path.Combine(args[0], "directory.orleans"), binary.Serialize(directory).ToArray());
File.WriteAllBytes(Path.Combine(args[0], "grants.orleans"), binary.Serialize(grants).ToArray());
File.WriteAllText(Path.Combine(args[0], "directory.json"), json.Serialize(directory).ToString());
File.WriteAllText(Path.Combine(args[0], "grants.json"), json.Serialize(grants).ToString());
Console.WriteLine("Captured legacy identity storage payloads.");
