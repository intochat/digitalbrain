using DigitalBrain.Apps;
using Microsoft.Extensions.DependencyInjection;
using Orleans.Serialization;
using Orleans.Storage;

namespace DigitalBrain.Modules.Apps.Tests.Unit;

public sealed class LegacyCaptureFacts
{
    [Fact]
    public async Task Capture()
    {
        await using var brain = await PackageBrain.StartAsync(TestContext.Current.CancellationToken);
        var serializer = new OrleansGrainStorageSerializer(brain.Brain.SiloServices.GetRequiredService<Serializer>());
        var state = new AppState { Status = AppStatus.Uninstalled, ProgramGeneration = 3, ScriptPaths = ["app.cs"] };
        await File.WriteAllBytesAsync(Path.Combine(AppContext.BaseDirectory, "app.orleans"), serializer.Serialize(state).ToArray(), TestContext.Current.CancellationToken);
    }
}
