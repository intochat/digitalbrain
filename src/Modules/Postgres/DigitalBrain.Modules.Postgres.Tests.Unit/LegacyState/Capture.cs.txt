using DigitalBrain.Postgres;
using Microsoft.Extensions.DependencyInjection;
using Orleans.Serialization;
using Orleans.Storage;

namespace DigitalBrain.Modules.Postgres.Tests.Unit;

public sealed class LegacyCaptureFacts
{
    [Fact]
    public async Task Capture()
    {
        await using var brain = await UnitTest.Create().WithModule<PostgresModule>()
            .ConfigureSilo(silo => silo.Configuration["ConnectionStrings:postgres"] = "Host=localhost;Database=sample;Username=reader")
            .StartAsync(TestContext.Current.CancellationToken);
        var serializer = new OrleansGrainStorageSerializer(brain.SiloServices.GetRequiredService<Serializer>());
        var state = new PostgresTableState
        {
            Owner = System.Text.Json.JsonSerializer.Serialize(new[] { DigitalBrain.Core.Enforcement.BrainScope.Create("alice", "brain").Id, "app" }),
            Accepted = new("legacy_table", new([new("id", "text")], ["id"]), 1)
        };
        await File.WriteAllBytesAsync(Path.Combine(AppContext.BaseDirectory, "table.orleans"), serializer.Serialize(state).ToArray(), TestContext.Current.CancellationToken);
    }
}
